from __future__ import annotations

from dataclasses import dataclass

import torch
import torch.nn as nn


@dataclass(frozen=True)
class MixedTypeModelConfig:
    categorical_cardinalities: list[int]
    numerical_dim: int
    binary_dim: int = 1
    max_events: int = 512
    d_model: int = 64
    num_heads: int = 4
    feature_layers: int = 1
    temporal_layers: int = 2
    ff_dim: int = 128
    dropout: float = 0.2


class SelfAttentionBlock(nn.Module):
    def __init__(
        self,
        *,
        d_model: int,
        num_heads: int,
        ff_dim: int,
        dropout: float,
    ):
        super().__init__()

        if d_model % num_heads != 0:
            raise ValueError(
                "d_model must be divisible by num_heads"
            )

        self.norm1 = nn.LayerNorm(d_model)
        self.attn = nn.MultiheadAttention(
            d_model,
            num_heads,
            dropout=dropout,
            batch_first=True,
        )
        self.attn_dropout = nn.Dropout(dropout)

        self.norm2 = nn.LayerNorm(d_model)
        self.ff = nn.Sequential(
            nn.Linear(d_model, ff_dim),
            nn.GELU(),
            nn.Dropout(dropout),
            nn.Linear(ff_dim, d_model),
            nn.Dropout(dropout),
        )

    def forward(
        self,
        x: torch.Tensor,
        *,
        key_padding_mask: torch.Tensor | None = None,
        return_attention: bool = False,
    ):
        y = self.norm1(x)

        attended, weights = self.attn(
            y,
            y,
            y,
            key_padding_mask=key_padding_mask,
            need_weights=return_attention,
            average_attn_weights=False,
        )

        x = x + self.attn_dropout(attended)
        x = x + self.ff(self.norm2(x))

        return x, weights


class MixedTypeEventEncoder(nn.Module):
    TYPE_CAT = 0
    TYPE_NUM = 1
    TYPE_BIN = 2
    TYPE_DELTA = 3

    def __init__(
        self,
        *,
        categorical_cardinalities: list[int],
        numerical_dim: int,
        binary_dim: int,
        d_model: int,
        num_heads: int,
        num_layers: int,
        ff_dim: int,
        dropout: float,
    ):
        super().__init__()

        self.categorical_dim = len(categorical_cardinalities)
        self.numerical_dim = int(numerical_dim)
        self.binary_dim = int(binary_dim)

        if self.binary_dim != 1:
            raise ValueError(
                "Current baseline expects one binary field."
            )

        self.cat_embeddings = nn.ModuleList([
            nn.Embedding(int(cardinality), d_model)
            for cardinality in categorical_cardinalities
        ])

        self.num_weight = nn.Parameter(
            torch.empty(self.numerical_dim, d_model)
        )
        self.num_bias = nn.Parameter(
            torch.zeros(self.numerical_dim, d_model)
        )
        self.num_missing_vector = nn.Parameter(
            torch.zeros(self.numerical_dim, d_model)
        )

        self.binary_embedding = nn.Embedding(2, d_model)

        self.delta_weight = nn.Parameter(
            torch.empty(d_model)
        )
        self.delta_bias = nn.Parameter(
            torch.zeros(d_model)
        )

        self.total_feature_tokens = (
            self.categorical_dim
            + self.numerical_dim
            + self.binary_dim
            + 1
        )

        self.field_embedding = nn.Embedding(
            self.total_feature_tokens,
            d_model,
        )
        self.type_embedding = nn.Embedding(4, d_model)

        field_type_ids = (
            [self.TYPE_CAT] * self.categorical_dim
            + [self.TYPE_NUM] * self.numerical_dim
            + [self.TYPE_BIN] * self.binary_dim
            + [self.TYPE_DELTA]
        )

        self.register_buffer(
            "field_type_ids",
            torch.tensor(field_type_ids, dtype=torch.long),
            persistent=False,
        )

        self.blocks = nn.ModuleList([
            SelfAttentionBlock(
                d_model=d_model,
                num_heads=num_heads,
                ff_dim=ff_dim,
                dropout=dropout,
            )
            for _ in range(num_layers)
        ])

        self.output_norm = nn.LayerNorm(d_model)

        nn.init.normal_(self.num_weight, mean=0.0, std=0.02)
        nn.init.normal_(
            self.num_missing_vector,
            mean=0.0,
            std=0.02,
        )
        nn.init.normal_(self.delta_weight, mean=0.0, std=0.02)

    def build_feature_tokens(
        self,
        *,
        x_cat: torch.Tensor,
        x_num: torch.Tensor,
        x_num_missing: torch.Tensor,
        x_bin: torch.Tensor,
        delta_t_log1p: torch.Tensor,
    ) -> torch.Tensor:
        cat_tokens = torch.stack(
            [
                emb(x_cat[:, i])
                for i, emb in enumerate(self.cat_embeddings)
            ],
            dim=1,
        )

        num_tokens = (
            x_num.unsqueeze(-1)
            * self.num_weight.unsqueeze(0)
            + self.num_bias.unsqueeze(0)
        )

        num_tokens = (
            num_tokens
            + x_num_missing.unsqueeze(-1)
            * self.num_missing_vector.unsqueeze(0)
        )

        bin_token = self.binary_embedding(
            x_bin[:, 0].long().clamp(0, 1)
        ).unsqueeze(1)

        delta_token = (
            delta_t_log1p[:, 0].unsqueeze(-1)
            * self.delta_weight.unsqueeze(0)
            + self.delta_bias.unsqueeze(0)
        ).unsqueeze(1)

        tokens = torch.cat(
            [
                cat_tokens,
                num_tokens,
                bin_token,
                delta_token,
            ],
            dim=1,
        )

        field_ids = torch.arange(
            self.total_feature_tokens,
            device=tokens.device,
        )

        tokens = (
            tokens
            + self.field_embedding(field_ids)[None, :, :]
            + self.type_embedding(
                self.field_type_ids
            )[None, :, :]
        )

        return tokens

    def forward(
        self,
        *,
        x_cat: torch.Tensor,
        x_num: torch.Tensor,
        x_num_missing: torch.Tensor,
        x_bin: torch.Tensor,
        delta_t_log1p: torch.Tensor,
        return_attention: bool = False,
    ):
        tokens = self.build_feature_tokens(
            x_cat=x_cat,
            x_num=x_num,
            x_num_missing=x_num_missing,
            x_bin=x_bin,
            delta_t_log1p=delta_t_log1p,
        )

        last_attention = None

        for i, block in enumerate(self.blocks):
            tokens, weights = block(
                tokens,
                return_attention=(
                    return_attention
                    and i == len(self.blocks) - 1
                ),
            )
            if weights is not None:
                last_attention = weights

        tokens = self.output_norm(tokens)

        # Feature-level classification-style pooling.
        event_embedding = tokens.mean(dim=1)

        return event_embedding, last_attention


class MixedTypeSelfAttentionClassifier(nn.Module):
    def __init__(
        self,
        config: MixedTypeModelConfig,
    ):
        super().__init__()
        self.config = config

        self.event_encoder = MixedTypeEventEncoder(
            categorical_cardinalities=
                config.categorical_cardinalities,
            numerical_dim=
                config.numerical_dim,
            binary_dim=
                config.binary_dim,
            d_model=
                config.d_model,
            num_heads=
                config.num_heads,
            num_layers=
                config.feature_layers,
            ff_dim=
                config.ff_dim,
            dropout=
                config.dropout,
        )

        self.position_embedding = nn.Embedding(
            config.max_events,
            config.d_model,
        )

        self.temporal_blocks = nn.ModuleList([
            SelfAttentionBlock(
                d_model=config.d_model,
                num_heads=config.num_heads,
                ff_dim=config.ff_dim,
                dropout=config.dropout,
            )
            for _ in range(config.temporal_layers)
        ])

        self.temporal_norm = nn.LayerNorm(config.d_model)

        self.classifier = nn.Sequential(
            nn.Linear(config.d_model, config.d_model),
            nn.GELU(),
            nn.Dropout(config.dropout),
            nn.Linear(config.d_model, 1),
        )

    def forward(
        self,
        *,
        x_cat: torch.Tensor,
        x_num: torch.Tensor,
        x_num_missing: torch.Tensor,
        x_bin: torch.Tensor,
        delta_t_log1p: torch.Tensor,
        attention_mask: torch.Tensor,
        return_attention: bool = False,
    ):
        b, t, _ = x_cat.shape

        if t > self.config.max_events:
            raise ValueError(
                f"T={t} exceeds max_events={self.config.max_events}"
            )

        valid = attention_mask.bool()
        flat_valid = valid.reshape(-1)

        flat_cat = x_cat.reshape(
            b * t,
            x_cat.shape[-1],
        )[flat_valid]

        flat_num = x_num.reshape(
            b * t,
            x_num.shape[-1],
        )[flat_valid]

        flat_missing = x_num_missing.reshape(
            b * t,
            x_num_missing.shape[-1],
        )[flat_valid]

        flat_bin = x_bin.reshape(
            b * t,
            x_bin.shape[-1],
        )[flat_valid]

        flat_delta = delta_t_log1p.reshape(
            b * t,
            1,
        )[flat_valid]

        event_valid, feature_attention = self.event_encoder(
            x_cat=flat_cat,
            x_num=flat_num,
            x_num_missing=flat_missing,
            x_bin=flat_bin,
            delta_t_log1p=flat_delta,
            return_attention=return_attention,
        )

        event_embeddings = torch.zeros(
            b * t,
            self.config.d_model,
            device=x_cat.device,
            dtype=event_valid.dtype,
        )

        event_embeddings[flat_valid] = event_valid

        event_embeddings = event_embeddings.view(
            b,
            t,
            self.config.d_model,
        )

        positions = torch.arange(
            t,
            device=x_cat.device,
        )

        temporal = (
            event_embeddings
            + self.position_embedding(positions)[None, :, :]
        )

        key_padding_mask = ~valid
        last_temporal_attention = None

        for i, block in enumerate(self.temporal_blocks):
            temporal, weights = block(
                temporal,
                key_padding_mask=key_padding_mask,
                return_attention=(
                    return_attention
                    and i == len(self.temporal_blocks) - 1
                ),
            )
            if weights is not None:
                last_temporal_attention = weights

        temporal = self.temporal_norm(temporal)

        mask_float = valid.unsqueeze(-1).to(temporal.dtype)

        pooled = (
            (temporal * mask_float).sum(dim=1)
            / mask_float.sum(dim=1).clamp_min(1.0)
        )

        logits = self.classifier(pooled).squeeze(-1)

        if return_attention:
            return {
                "logits": logits,
                "feature_attention": feature_attention,
                "temporal_attention": last_temporal_attention,
                "valid_event_mask": valid,
            }

        return logits
